Practical Examples ​

1. Auto Height ​

javascript
import { BottomSheetManager } from 'bottom-sheet-manager';

const bs = new BottomSheetManager();

bs.open({
  title: 'Auto Height',
  content: '<p>Content with dynamic height</p>'
});

2. Fixed Height ​

javascript
bs.open({
  title: 'Fixed Height',
  height: '400px',
  content: '<p>Fixed height of 400 pixels</p>'
});

3. Snap Points ​

javascript
bs.open({
  title: 'Snap Points',
  height: '350px',
  snapOnDrag: true,
  content: '<p>Draggable between snap positions</p>'
});

4. Start Expanded ​

javascript
bs.open({
  title: 'Start Expanded',
  height: '350px',
  snapOnDrag: true,
  startExpanded: true,
  content: '<p>Opens initially at max height</p>'
});

5. Scrollable Content ​

javascript
bs.open({
  title: 'Scrollable Content',
  height: '300px',
  content: `
    <div style="height: 600px;">
      <p>Top of content</p>
      <p style="margin-top: 500px;">Bottom of content</p>
    </div>
  `
});

6. Nested Bottom Sheets ​

javascript
bs.open({
  title: 'First Bottom Sheet',
  height: '300px',
  content: '<button id="btn-child">Open Second Sheet</button>',
  onOpen: () => {
    document.getElementById('btn-child').onclick = () => {
      bs.open({
        title: 'Second Bottom Sheet',
        height: '200px',
        content: '<p>Top layer bottom sheet</p>'
      });
    };
  }
});
